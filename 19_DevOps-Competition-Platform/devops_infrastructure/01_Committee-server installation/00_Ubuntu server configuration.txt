# การทำ NAT (Network Address Translation) แชร์อินเตอร์เน็ตให้เครื่อข่ายลูก
# ตรวจสอบสถานะ ip_forward:

# การสร้าง:

# อนุญาตให้ข้อมูลที่ตอบกลับมาวิ่งเข้าได้
iptables -A FORWARD -m state --state RELATED,ESTABLISHED -j ACCEPT
# อนุญาตให้เครื่องทั้งหมดใน Interface ส่งข้อมูลออกไปได้ (เช่น enp2s0):
iptables -A FORWARD -i enp2s0 -j ACCEPT

# อนุญาตให้ IP เครื่องใน Interface ส่งข้อมูลออกไปได้ (เช่น enp2s0: 192.168.1.2):
iptables -t nat -A POSTROUTING -o enp3s0 -j MASQUERADE
iptables -A FORWARD -m state --state RELATED,ESTABLISHED -j ACCEPT
iptables -A FORWARD -i enp2s0 -s 192.168.1.2 -j ACCEPT
iptables -A FORWARD -i enp2s0 -j DROP

# กรณีทำ WiFi to Lan Sharing
# 1. Sharing Setup
Press Win + R -> ncpa.cpl -> Right-click your Wi-Fi adapter -> Properties -> Sharing tab
  Check: "Allow other network users to connect..."
  Dropdown: Select your Ethernet / LAN adapter from the list

# 2. Verify the IP was Actually Assigned
ipconfig
  Ethernet adapter show: 192.168.137.1

# 3. Turn off Windows Firewall
Start menu -> Windows Defender Firewall -> Click "Turn Windows Defender Firewall on or off"
   Check: turn off the firewall for both Private and Public networks.

# 4. Restart the ICS Service
Press Win + R -> services.msc -> find "Internet Connection Sharing (ICS)" -> Right-click and select "Restart"
