# Pull & Tag & Push all images
# สำหรับ Windows | PowerShell เวอร์ชั่นเก่า (5.1)
Get-Content images.txt | ForEach-Object { 
    docker pull $_; 
    docker tag $_ "127.0.0.1:5000/$_"; 
    docker push "127.0.0.1:5000/$_" 
}

# สำหรับ Linux/macOS/WSL
while read img; do docker pull "$img" && docker tag "$img" "127.0.0.1:5000/$img" && docker push "127.0.0.1:5000/$img"; done < images.txt

# Delete image prefix tags (Delete all registry container & volume):
docker rmi $(docker images --format "{{.Repository}}:{{.Tag}}"|findstr "127.0.0.1:5000/")
