docker volume create nginx_logs

mkdir -p ~/docker/nginx-stack/data && cd ~/docker/nginx-stack
nano docker-compose.yml

Add .yml stacks[name="nginx"]:

networks:
  app_net:
    external: true

volumes:
  nginx_logs:
    external: true

services:
  nginx:
    image: nginx:latest
    container_name: nginx
    restart: always
    networks:
      - app_net
    ports:
      - "80:80"      # เชื่อมพอร์ต 80 ของเครื่องโฮสต์ เข้ากับพอร์ต 80 ของ Container
    volumes:
      - nginx_logs:/var/log/nginx
      - ./nginx/conf.d:/etc/nginx/conf.d
      - ./html:/usr/share/nginx/html:ro

docker compose up -d

ufw allow 80/tcp
ufw reload
ufw status
