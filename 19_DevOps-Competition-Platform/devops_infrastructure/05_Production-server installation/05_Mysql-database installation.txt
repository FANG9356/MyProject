docker volume create mysql_data

mkdir -p ~/docker/mysql-stack/data && cd ~/docker/mysql-stack
nano docker-compose.yml

Add .yml stacks[name="mysql"]:

networks:
  app_net:
    external: true

volumes:
  mysql_data:
    external: true

services:
  mysql:
    image: mysql:latest
    container_name: mysql
    restart: always
    networks:
      - app_net
    ports:
      - "3306:3306"
    environment:
      - TZ=Asia/Bangkok
      - MYSQL_ROOT_PASSWORD=11111111
      - MYSQL_DATABASE=devops_iot_db
    volumes:
      - mysql_data:/var/lib/mysql

docker compose up -d
