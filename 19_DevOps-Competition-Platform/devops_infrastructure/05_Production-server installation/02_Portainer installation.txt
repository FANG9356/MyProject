install type 1:
install:
	docker run -d -p 9443:9443 
	--name portainer 
	--restart=always 
	-v /var/run/docker.sock:/var/run/docker.sock 
	-v portainer_data:/data 
	portainer/portainer-ce:latest

install type 2:
docker network create app_net
docker volume create portainer_data

mkdir -p ~/docker/portainer-stack/data && cd ~/docker/portainer-stack
nano docker-compose.yml

Add .yml stacks[name="portainer "]:

networks:
  app_net:
    external: true

volumes:
  portainer_data:
    external: true

services:
  portainer:
    image: portainer/portainer-ce:latest
    container_name: portainer
    restart: always
    networks:
      - app_net
    ports:
      - "9443:9443"
    volumes:
      - /var/run/docker.sock:/var/run/docker.sock
      - portainer_data:/data

docker compose up -d

check all container:
	docker ps -a
start container:
	docker start [CONTAINER_ID / Name]
restart container:
	docker restart [CONTAINER_ID / Name]
stop container:
	docker stop [CONTAINER_ID / Name]
delete container:
	docker rm [CONTAINER_ID / Name]
delete all container:
	docker container prune

check all image:
	docker images
delete image:
	docker rmi [IMAGE_Name]

list all volume:
	docker volume ls
