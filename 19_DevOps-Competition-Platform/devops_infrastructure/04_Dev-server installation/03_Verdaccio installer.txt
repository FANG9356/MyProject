docker run -d --name verdaccio -p 4873:4873 verdaccio/verdaccio

docker volume create verdaccio_storage
docker volume create verdaccio_conf

mkdir -p ~/docker/verdaccio-stack/data && cd ~/docker/verdaccio-stack
nano docker-compose.yml

Add .yml stacks[name=["verdaccio"]]:

networks:
  app_net:
    external: true

volumes:
  verdaccio_storage:
    external: true

  verdaccio_conf:
    external: true

services:
  verdaccio:
    image: 192.168.110.242:8080/verdaccio/verdaccio:latest
    container_name: verdaccio
    restart: always
    networks:
      - app_net
    ports:
      - "4873:4873"
    environment:
      HTTP_PROXY: http://192.168.87.242:3142
      HTTPS_PROXY: http://192.168.87.242:3142
    volumes:
      - verdaccio_storage:/verdaccio/storage
      - verdaccio_conf:/verdaccio/conf

docker compose up -d
