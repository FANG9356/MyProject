# Step 1: Create EMQX connection:
network group -> mqtt in -> config:
  Create new server: 
    Name: EMQX Broker
    Server: emqx (Container name)
    Port: 1883
    Protocol: MQTT V5, V3 (Match with EMQX WebSocket)

common group -> Debug:

# Link 
  "mqtt in" to "debug"

# Deploy: 
  "mqtt in": show "connected".
  "debug": show values from EMQX.



# Step 2: Prepare function to insert data:
function group -> Function:
  Name: insert db
  On Message: (JS coding): 

// Get JSON data from MQTT 
let device = "ESP32_Node_A";
let temp = msg.payload.temperature;
let press = msg.payload.pressure;

// SQL Command
msg.topic = 
"INSERT INTO bmp280_data "+
"(device_id, temperature, pressure) "+
"VALUES (?, ?, ?)";

// Addind array value sent to (?,?)MySQL  
msg.payload = [device, temp, press];

return msg;



# Link 
  "mqtt in" to "function", and "function" to "debug"

# Deploy: 
  "debug": show array data to prepare inserting data.
